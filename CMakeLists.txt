cmake_minimum_required(VERSION 3.16)
project(MultiLogAnalyzer VERSION 1.0.0 LANGUAGES CXX)

option(MULTILOG_ENABLE_ASAN "Enable AddressSanitizer" OFF)
option(MULTILOG_ENABLE_TSAN "Enable ThreadSanitizer" OFF)
option(MULTILOG_BUILD_TESTS "Build tests (GoogleTest is fetched by CMake)" ON)
if(MULTILOG_ENABLE_ASAN AND MULTILOG_ENABLE_TSAN)
  message(FATAL_ERROR "AddressSanitizer and ThreadSanitizer cannot be enabled together")
endif()
find_package(Threads REQUIRED)
add_library(multilog_core src/LogParser.cpp src/ThreadPool.cpp src/LogAnalyzer.cpp src/SuspiciousActivityDetector.cpp src/ReportGenerator.cpp src/BenchmarkManager.cpp)
target_include_directories(multilog_core PUBLIC include)
target_compile_features(multilog_core PUBLIC cxx_std_17)
target_link_libraries(multilog_core PUBLIC Threads::Threads)
if(MSVC)
  target_compile_options(multilog_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(multilog_core PRIVATE -Wall -Wextra -Wpedantic)
  if(MULTILOG_ENABLE_ASAN)
    target_compile_options(multilog_core PUBLIC -fsanitize=address -fno-omit-frame-pointer)
    target_link_options(multilog_core PUBLIC -fsanitize=address)
  elseif(MULTILOG_ENABLE_TSAN)
    target_compile_options(multilog_core PUBLIC -fsanitize=thread -fno-omit-frame-pointer)
    target_link_options(multilog_core PUBLIC -fsanitize=thread)
  endif()
endif()
add_executable(multilog src/main.cpp)
target_link_libraries(multilog PRIVATE multilog_core)
if(MSVC)
  target_compile_options(multilog PRIVATE /W4 /permissive-)
else()
  target_compile_options(multilog PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(MULTILOG_BUILD_TESTS)
  include(FetchContent)
  FetchContent_Declare(googletest URL https://github.com/google/googletest/archive/refs/tags/v1.14.0.zip)
  FetchContent_MakeAvailable(googletest)
  enable_testing()
  add_executable(multilog_tests tests/test_multilog.cpp)
  target_link_libraries(multilog_tests PRIVATE multilog_core GTest::gtest_main)
  include(GoogleTest)
  gtest_discover_tests(multilog_tests)
endif()
